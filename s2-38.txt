function frame(ts){
 D.frames=(D.frames||0)+1;
 if(D.mode!=='play'||D._notick){D.raf=0;return;}
 D.raf=requestAnimationFrame(frame);
 if(!D.last)D.last=ts;
 D.acc+=Math.min(100,ts-D.last);D.last=ts;
 while(D.acc>=16.666){D.acc-=16.666;update();}
 draw();
}
